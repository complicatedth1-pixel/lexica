-- Books/library metadata table
create table public.books (
  id text not null,
  user_id uuid not null references auth.users(id) on delete cascade,
  name text not null default '',
  tree_data jsonb not null default '[]',
  highlights jsonb not null default '{}',
  notes jsonb not null default '{}',
  last_opened bigint not null default 0,
  is_pdf boolean not null default false,
  is_pdf_viewer boolean not null default false,
  pdf_num_pages integer,
  page_times jsonb not null default '{}',
  pdf_highlights jsonb not null default '{}',
  created_at timestamptz not null default now(),
  primary key (id, user_id)
);

-- RLS
alter table public.books enable row level security;
create policy "Users see own books" on public.books for all using (auth.uid() = user_id);

-- Storage bucket for PDFs
insert into storage.buckets (id, name, public) values ('pdfs', 'pdfs', false);

-- Storage RLS: users can only access their own folder
create policy "User pdf upload" on storage.objects for insert
  with check (bucket_id = 'pdfs' and (storage.foldername(name))[1] = auth.uid()::text);

create policy "User pdf read" on storage.objects for select
  using (bucket_id = 'pdfs' and (storage.foldername(name))[1] = auth.uid()::text);

create policy "User pdf delete" on storage.objects for delete
  using (bucket_id = 'pdfs' and (storage.foldername(name))[1] = auth.uid()::text);

-- Lexica — Prompt Presets Schema
-- Run this in your Supabase SQL editor
-- ─────────────────────────────────────────────────────────────

-- 1. Named prompt presets (one row per preset per user)
create table public.prompt_presets (
  id          uuid primary key default gen_random_uuid(),
  user_id     uuid not null references auth.users(id) on delete cascade,
  type        text not null check (type in ('index', 'page')),
  name        text not null default 'New Preset',
  instructions text not null default '',
  created_at  timestamptz not null default now()
);

alter table public.prompt_presets enable row level security;
create policy "Users manage own presets"
  on public.prompt_presets for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);

create index on public.prompt_presets (user_id, type);

-- ─────────────────────────────────────────────────────────────

-- 2. Per-user settings (one row per user, tracks active preset)
create table public.user_settings (
  user_id               uuid primary key references auth.users(id) on delete cascade,
  active_index_preset_id uuid references public.prompt_presets(id) on delete set null,
  active_page_preset_id  uuid references public.prompt_presets(id) on delete set null,
  updated_at            timestamptz not null default now()
);

alter table public.user_settings enable row level security;
create policy "Users manage own settings"
  on public.user_settings for all
  using (auth.uid() = user_id)
  with check (auth.uid() = user_id);
